PROCESSAR A

<?php
if ($_SERVER['REQUEST_METHOD'] == 'GET'){

    $nombre = $_GET["nombre"];
    $apellido = $_GET["apellido"];
    $edad = $_GET["edad"];
    $direccion = $_GET["direccion"];
    $telefono = $_GET["telefono"];
    $email = $_GET["correo"];
    $genero = $_GET["genero"];
    $idiomas = $_GET["idiomes"] ?? [];

    if (empty($nombre)) {
        echo "El 'Nombre' es obligatorio.<br>";
    }

    if (empty($apellido)) {
        echo "El 'Apellido' es obligatorio.<br>";
    }

    if (empty($edad)) {
        echo "La 'Edad' es obligatoria.<br>";
    } elseif (!filter_var($edad, FILTER_VALIDATE_INT)) {
        echo "La edad no es válida.<br>";
    }

    if (empty($email)) {
        echo "El 'email' es obligatorio.<br>";
    } elseif (!filter_var($email, FILTER_VALIDATE_EMAIL)) {
        echo "El email no es válido.<br>";
    }

    if (!empty($idiomas)) {
        foreach ($idiomas as $idioma) {
            if ($idioma == "Español") {
                echo "Bienvenido!<br>";
            } elseif ($idioma == "Inglés") {
                echo "Welcome!<br>";
            } elseif ($idioma == "Francés") {
                echo "Bienvenue!<br>";
            } elseif ($idioma == "Alemán") {
                echo "Willkommen!<br>";
            } elseif ($idioma == "Italiano") {
                echo "Benvenuto!<br>";
            }
        }
    } else {
        echo "No se ha seleccionado ningún idioma.<br>";
    }

    echo "<br><strong>Datos introducidos:</strong><br>";
    echo "Nombre: $nombre<br>";
    echo "Apellido: $apellido<br>";
    echo "Edad: $edad<br>";
    echo "Dirección: $direccion<br>";
    echo "Teléfono: $telefono<br>";
    echo "Email: $email<br>";
    echo "Género: $genero<br><br>";

    echo "Idiomas: " . implode(", ", $idiomas) . "<br><br>";
}
?>

<?php
if (isset($_FILES['fichero'])) {
    $nombre = $_FILES['fichero']['name'];
    $tmp = $_FILES['fichero']['tmp_name'];
    $destino = '/var/www/html/ficheros/' . $nombre;

    if (move_uploaded_file($tmp, $destino)) {
        echo "<p>El fichero '$nombre' se ha guardado correctamente.</p>";
        echo "<h3>El contenido del fichero:</h3>";
        echo "<pre>" . htmlspecialchars(file_get_contents($destino)) . "</pre>";
    } else {
        echo "<p>Error de guardado.</p>";
    }
} else {
    echo "<p>No se ha enviado ningún fichero</p>";
}
?>
